    <script>
        const audioContext = new (window.AudioContext || window.webkitAudioContext)();
        
        const notes = ['C', 'C#', 'D', 'D#', 'E', 'F', 'F#', 'G', 'G#', 'A', 'A#', 'B'];
        const octaves = [3, 4, 5];
        
        const chords = {
            'C Major': ['C4', 'E4', 'G4'],
            'D Major': ['D4', 'F#4', 'A4'],
            'E Major': ['E4', 'G#4', 'B4'],
            'F Major': ['F4', 'A4', 'C5'],
            'G Major': ['G4', 'B4', 'D5'],
            'A Major': ['A4', 'C#5', 'E5'],
            'B Major': ['B4', 'D#5', 'F#5'],
            'C Minor': ['C4', 'D#4', 'G4'],
            'D Minor': ['D4', 'F4', 'A4'],
            'E Minor': ['E4', 'G4', 'B4'],
            'F Minor': ['F4', 'G#4', 'C5'],
            'G Minor': ['G4', 'A#4', 'D5'],
            'A Minor': ['A4', 'C5', 'E5'],
            'B Minor': ['B4', 'D5', 'F#5']
        };

        let currentMode = 'single';
        let currentAnswer = [];
        let correctCount = 0;
        let totalCount = 0;

        function noteToFrequency(note) {
            const noteMatch = note.match(/([A-G]#?)(\d)/);
            if (!noteMatch) return 0;
            
            const noteName = noteMatch[1];
            const octave = parseInt(noteMatch[2]);
            
            const noteIndex = notes.indexOf(noteName);
            const semitone = (octave - 4) * 12 + noteIndex;
            return 440 * Math.pow(2, (semitone - 9) / 12);
        }

        function playNote(frequency, startTime = 0, duration = 1) {
            const oscillator = audioContext.createOscillator();
            const gainNode = audioContext.createGain();
            
            oscillator.connect(gainNode);
            gainNode.connect(audioContext.destination);
            
            oscillator.frequency.value = frequency;
            oscillator.type = 'sine';
            
            const now = audioContext.currentTime;
            gainNode.gain.setValueAtTime(0, now + startTime);
            gainNode.gain.linearRampToValueAtTime(0.3, now + startTime + 0.01);
            gainNode.gain.exponentialRampToValueAtTime(0.01, now + startTime + duration);
            
            oscillator.start(now + startTime);
            oscillator.stop(now + startTime + duration);
        }

        function generateQuestion() {
            currentAnswer = [];
            
            if (currentMode === 'single') {
                const note = notes[Math.floor(Math.random() * notes.length)];
                const octave = octaves[Math.floor(Math.random() * octaves.length)];
                currentAnswer = [note + octave];
            } else if (currentMode === 'double') {
                for (let i = 0; i < 2; i++) {
                    const note = notes[Math.floor(Math.random() * notes.length)];
                    const octave = octaves[Math.floor(Math.random() * octaves.length)];
                    currentAnswer.push(note + octave);
                }
            } else if (currentMode === 'triple') {
                for (let i = 0; i < 3; i++) {
                    const note = notes[Math.floor(Math.random() * notes.length)];
                    const octave = octaves[Math.floor(Math.random() * octaves.length)];
                    currentAnswer.push(note + octave);
                }
            } else if (currentMode === 'chord') {
                const chordNames = Object.keys(chords);
                const randomChord = chordNames[Math.floor(Math.random() * chordNames.length)];
                currentAnswer = [randomChord];
            }
        }

        function playSound() {
            if (currentMode === 'chord') {
                const chordName = currentAnswer[0];
                const chordNotes = chords[chordName];
                chordNotes.forEach(note => {
                    playNote(noteToFrequency(note), 0, 2);
                });
            } else {
                currentAnswer.forEach((note, index) => {
                    playNote(noteToFrequency(note), index * 1.2, 1);
                });
            }
        }

        function setupInputs() {
            const inputArea = document.getElementById('inputArea');
            inputArea.innerHTML = '';
            
            if (currentMode === 'chord') {
                inputArea.innerHTML = `
                    <div class="input-group">
                        <input type="text" id="input0" placeholder="Chord name (e.g., C Major, A Minor)">
                        <button class="submit-btn" onclick="checkAnswer()">Submit</button>
                    </div>
                `;
            } else {
                const count = currentMode === 'single' ? 1 : currentMode === 'double' ? 2 : 3;
                let html = '<div class="input-group">';
                for (let i = 0; i < count; i++) {
                    html += `<input type="text" id="input${i}" placeholder="e.g., A#3">`;
                }
                html += '<button class="submit-btn" onclick="checkAnswer()">Submit</button></div>';
                inputArea.innerHTML = html;
            }

            // Add enter key support
            inputArea.querySelectorAll('input').forEach((input, index) => {
                input.addEventListener('keypress', (e) => {
                    if (e.key === 'Enter') {
                        checkAnswer();
                    }
                });
            });
        }

        function checkAnswer() {
            const resultDiv = document.getElementById('result');
            let userAnswer = [];
            
            if (currentMode === 'chord') {
                userAnswer = [document.getElementById('input0').value.trim()];
            } else {
                const count = currentMode === 'single' ? 1 : currentMode === 'double' ? 2 : 3;
                for (let i = 0; i < count; i++) {
                    userAnswer.push(document.getElementById(`input${i}`).value.trim().toUpperCase());
                }
            }

            totalCount++;
            let isCorrect = false;

            if (currentMode === 'chord') {
                isCorrect = userAnswer[0].toLowerCase() === currentAnswer[0].toLowerCase();
            } else {
                isCorrect = JSON.stringify(userAnswer.sort()) === JSON.stringify(currentAnswer.sort());
            }

            if (isCorrect) {
                correctCount++;
                resultDiv.className = 'result correct';
                resultDiv.innerHTML = '✓ Correct!';
            } else {
                resultDiv.className = 'result incorrect';
                resultDiv.innerHTML = `✗ Incorrect<div class="answer-display">Correct answer: ${currentAnswer.join(', ')}</div>`;
            }

            updateStats();
            
            setTimeout(() => {
                generateQuestion();
                setupInputs();
                resultDiv.innerHTML = '';
                resultDiv.className = 'result';
            }, 2000);
        }

        function updateStats() {
            document.getElementById('correctCount').textContent = correctCount;
            document.getElementById('totalCount').textContent = totalCount;
            const accuracy = totalCount > 0 ? Math.round((correctCount / totalCount) * 100) : 0;
            document.getElementById('accuracy').textContent = accuracy + '%';
        }

        // Event listeners
        document.querySelectorAll('.mode-btn').forEach(btn => {
            btn.addEventListener('click', () => {
                document.querySelectorAll('.mode-btn').forEach(b => b.classList.remove('active'));
                btn.classList.add('active');
                currentMode = btn.dataset.mode;
                generateQuestion();
                setupInputs();
                document.getElementById('result').innerHTML = '';
                document.getElementById('result').className = 'result';
            });
        });

        document.getElementById('playBtn').addEventListener('click', playSound);

        // Initialize
        generateQuestion();
        setupInputs();
    </script>
